4. parser.py

Ce fichier récupère les infos de base d’un site.

import re
from bs4 import BeautifulSoup
from urllib.parse import urljoin, urlparse


KEY_PAGES = [
    "/about",
    "/a-propos",
    "/contact",
    "/services",
    "/products",
    "/pricing",
    "/faq",
    "/legal",
    "/privacy",
    "/terms",
]


def clean_text(text: str) -> str:
    if not text:
        return ""
    text = re.sub(r"\s+", " ", text).strip()
    return text


def get_domain(url: str) -> str:
    parsed = urlparse(url)
    return parsed.netloc.lower().replace("www.", "")


def extract_title_and_meta(html: str) -> dict:
    soup = BeautifulSoup(html, "lxml")

    title = soup.title.string.strip() if soup.title and soup.title.string else ""

    meta_description = ""
    meta_desc_tag = soup.find("meta", attrs={"name": "description"})
    if meta_desc_tag and meta_desc_tag.get("content"):
        meta_description = meta_desc_tag["content"].strip()

    og_title = ""
    og_title_tag = soup.find("meta", attrs={"property": "og:title"})
    if og_title_tag and og_title_tag.get("content"):
        og_title = og_title_tag["content"].strip()

    og_description = ""
    og_desc_tag = soup.find("meta", attrs={"property": "og:description"})
    if og_desc_tag and og_desc_tag.get("content"):
        og_description = og_desc_tag["content"].strip()

    canonical = ""
    canonical_tag = soup.find("link", attrs={"rel": "canonical"})
    if canonical_tag and canonical_tag.get("href"):
        canonical = canonical_tag["href"].strip()

    hreflang_present = bool(soup.find("link", attrs={"hreflang": True}))

    return {
        "title": title,
        "meta_description": meta_description,
        "og_title": og_title,
        "og_description": og_description,
        "canonical": canonical,
        "hreflang_present": hreflang_present,
    }


def extract_jsonld(html: str) -> list:
    soup = BeautifulSoup(html, "lxml")
    scripts = soup.find_all("script", attrs={"type": "application/ld+json"})
    payloads = []

    for script in scripts:
        raw = script.string or script.get_text() or ""
        raw = raw.strip()
        if raw:
            payloads.append(raw)

    return payloads


def extract_emails(text: str) -> list:
    if not text:
        return []
    emails = re.findall(r"[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}", text)
    return list(sorted(set(emails)))


def extract_phones(text: str) -> list:
    if not text:
        return []
    phones = re.findall(r"\+?\d[\d\s().-]{7,}\d", text)
    return list(sorted(set(p.strip() for p in phones)))


def extract_visible_text(html: str) -> str:
    soup = BeautifulSoup(html, "lxml")

    for tag in soup(["script", "style", "noscript"]):
        tag.decompose()

    text = soup.get_text(separator=" ")
    return clean_text(text)


def detect_key_pages(base_url: str) -> list:
    found_pages = []
    for path in KEY_PAGES:
        found_pages.append(urljoin(base_url, path))
    return found_pages
5. scraper.py

Ce fichier télécharge la home page, le sitemap, et quelques pages utiles.

import requests
from urllib.parse import urljoin
from parser import get_domain, detect_key_pages

HEADERS = {
    "User-Agent": "AYA-Bot/0.1 (+https://aya.local)"
}

TIMEOUT = 12


def fetch_url(url: str) -> dict:
    try:
        response = requests.get(url, headers=HEADERS, timeout=TIMEOUT, allow_redirects=True)
        return {
            "url": response.url,
            "status_code": response.status_code,
            "ok": response.ok,
            "text": response.text if response.ok else "",
        }
    except requests.RequestException:
        return {
            "url": url,
            "status_code": None,
            "ok": False,
            "text": "",
        }


def fetch_homepage(url: str) -> dict:
    return fetch_url(url)


def fetch_sitemap(base_url: str) -> dict:
    sitemap_url = urljoin(base_url, "/sitemap.xml")
    return fetch_url(sitemap_url)


def fetch_key_pages(base_url: str) -> list:
    results = []
    for page_url in detect_key_pages(base_url):
        page = fetch_url(page_url)
        if page["ok"]:
            results.append(page)
    return results


def normalize_url(url: str) -> str:
    url = url.strip()
    if not url:
        return ""
    if not url.startswith("http://") and not url.startswith("https://"):
        url = "https://" + url
    return url


def scrape_site(url: str) -> dict:
    base_url = normalize_url(url)
    homepage = fetch_homepage(base_url)
    sitemap = fetch_sitemap(base_url)
    key_pages = fetch_key_pages(base_url)

    return {
        "input_url": base_url,
        "canonical_domain": get_domain(homepage["url"] if homepage["url"] else base_url),
        "homepage": homepage,
        "sitemap": sitemap,
        "key_pages": key_pages,
    }
6. generator.py

Ce fichier transforme les données scrapées en objet AYA_PREINDEX + ASR_DERIVED.

import json
import os
from parser import (
    extract_title_and_meta,
    extract_jsonld,
    extract_visible_text,
    extract_emails,
    extract_phones,
)


def detect_entity_name(meta: dict, domain: str) -> str:
    for value in [meta.get("og_title", ""), meta.get("title", "")]:
        if value:
            return value[:120]
    return domain


def detect_offer_text(text: str) -> list:
    keywords = [
        "service",
        "services",
        "product",
        "products",
        "solution",
        "platform",
        "application",
        "app",
        "software",
        "consulting",
        "restaurant",
        "delivery",
        "menu",
    ]
    lower_text = text.lower()
    found = []
    for kw in keywords:
        if kw in lower_text:
            found.append(kw)
    return list(sorted(set(found)))


def estimate_aio_score(blocks: dict, has_jsonld: bool, has_sitemap: bool) -> int:
    score = 0

    weights = {
        "identite": 15,
        "offre": 25,
        "processus_methodes": 10,
        "engagements_conformite": 10,
        "indicateurs": 15,
        "contenus_pedagogiques": 10,
        "structure_technique": 15,
    }

    for block_name, weight in weights.items():
        if blocks[block_name]["present"]:
            score += weight

    if has_jsonld:
        score += 5
    if has_sitemap:
        score += 5

    return min(score, 100)


def build_record(scraped: dict) -> dict:
    homepage_html = scraped["homepage"]["text"]
    homepage_url = scraped["homepage"]["url"] or scraped["input_url"]

    meta = extract_title_and_meta(homepage_html) if homepage_html else {}
    text = extract_visible_text(homepage_html) if homepage_html else ""
    jsonld_payloads = extract_jsonld(homepage_html) if homepage_html else []

    all_text = text
    pages_scanned = [homepage_url]

    for page in scraped["key_pages"]:
        all_text += " " + extract_visible_text(page["text"])
        pages_scanned.append(page["url"])

    emails = extract_emails(all_text)
    phones = extract_phones(all_text)

    has_jsonld = len(jsonld_payloads) > 0
    has_sitemap = bool(scraped["sitemap"]["ok"])

    offer_keywords = detect_offer_text(all_text)

    blocks = {
        "identite": {
            "present": bool(meta.get("title") or emails or phones),
            "fields": {
                "name": detect_entity_name(meta, scraped["canonical_domain"]),
                "website": homepage_url,
                "email": emails[0] if emails else "",
                "phone": phones[0] if phones else "",
            },
            "evidence": [meta.get("title", "")[:100]],
        },
        "offre": {
            "present": len(offer_keywords) > 0,
            "fields": {
                "services_or_products_detected": offer_keywords,
            },
            "evidence": offer_keywords[:5],
        },
        "processus_methodes": {
            "present": ("how it works" in all_text.lower()) or ("comment ça marche" in all_text.lower()),
            "fields": {},
            "evidence": [],
        },
        "engagements_conformite": {
            "present": ("privacy" in all_text.lower()) or ("terms" in all_text.lower()) or ("cookies" in all_text.lower()),
            "fields": {},
            "evidence": [],
        },
        "indicateurs": {
            "present": False,
            "fields": {},
            "evidence": [],
        },
        "contenus_pedagogiques": {
            "present": ("faq" in all_text.lower()) or ("documentation" in all_text.lower()) or ("glossary" in all_text.lower()),
            "fields": {},
            "evidence": [],
        },
        "structure_technique": {
            "present": has_jsonld or has_sitemap or bool(meta.get("canonical")),
            "fields": {
                "has_jsonld": has_jsonld,
                "has_sitemap": has_sitemap,
                "canonical_present": bool(meta.get("canonical")),
                "hreflang_present": bool(meta.get("hreflang_present")),
            },
            "evidence": [],
        },
    }

    score = estimate_aio_score(blocks, has_jsonld, has_sitemap)

    record = {
        "version": "AYA-PREINDEX-1.0",
        "source": {
            "input_url": scraped["input_url"],
            "canonical_domain": scraped["canonical_domain"],
            "pages_scanned": pages_scanned,
            "structured_data_found": {
                "asr_found": False,
                "jsonld_found": has_jsonld,
                "microdata_found": False,
                "sitemap_found": has_sitemap,
            },
        },
        "entity": {
            "is_organization": True,
            "name": detect_entity_name(meta, scraped["canonical_domain"]),
            "website": homepage_url,
            "country": "",
            "city": "",
            "languages": [],
            "contacts": {
                "email": emails[0] if emails else "",
                "phone": phones[0] if phones else "",
                "address": "",
            },
        },
        "sector_detection": {
            "sector_macro_id": "",
            "confidence": 0,
            "evidence": [],
        },
        "aio_blocks": blocks,
        "aoi_readiness": {
            "blocks_present": [k for k, v in blocks.items() if v["present"]],
            "structured_data_types": ["jsonld"] if has_jsonld else [],
            "estimated_aio_score": score,
            "score_explanation": "Simple MVP score based on detected blocks and technical structure.",
            "confidence": 60 if has_jsonld else 35,
        },
        "extracted_structured_payloads": {
            "asr": None,
            "jsonld": jsonld_payloads,
            "microdata": [],
        },
        "asr_derived": {
            "asr_status": "ASR_DERIVED",
            "source": "AYA",
            "confidence": 0.6 if has_jsonld else 0.35,
        },
    }

    return record


def save_record(record: dict, folder: str = "data") -> str:
    os.makedirs(folder, exist_ok=True)
    domain = record["source"]["canonical_domain"].replace("/", "_")
    filepath = os.path.join(folder, f"{domain}.json")

    with open(filepath, "w", encoding="utf-8") as f:
        json.dump(record, f, ensure_ascii=False, indent=2)

    return filepath
7. run_pipeline.py

Ce fichier lance tout le pipeline.

from scraper import scrape_site
from generator import build_record, save_record


def load_domains(filepath: str = "domains.txt") -> list:
    with open(filepath, "r", encoding="utf-8") as f:
        return [line.strip() for line in f if line.strip()]


def main():
    domains = load_domains()

    for domain in domains:
        print(f"Scraping: {domain}")
        try:
            scraped = scrape_site(domain)
            record = build_record(scraped)
            filepath = save_record(record)
            print(f"Saved: {filepath}")
        except Exception as e:
            print(f"Error with {domain}: {e}")


if __name__ == "__main__":
    main()
8. api/main.py

Ce fichier crée l’API AYA.

import json
import os
from fastapi import FastAPI, HTTPException, Query

app = FastAPI(title="AYA API", version="0.1")


DATA_FOLDER = "data"


def load_all_records():
    records = []
    if not os.path.exists(DATA_FOLDER):
        return records

    for filename in os.listdir(DATA_FOLDER):
        if filename.endswith(".json"):
            path = os.path.join(DATA_FOLDER, filename)
            with open(path, "r", encoding="utf-8") as f:
                records.append(json.load(f))
    return records


@app.get("/")
def root():
    return {
        "name": "AYA API",
        "status": "ok",
        "endpoints": ["/search", "/entity/{domain}", "/asr/{domain}"],
    }


@app.get("/search")
def search(q: str = Query(..., min_length=1)):
    q_lower = q.lower()
    results = []

    for record in load_all_records():
        name = record.get("entity", {}).get("name", "").lower()
        domain = record.get("source", {}).get("canonical_domain", "").lower()
        offer = json.dumps(record.get("aio_blocks", {}).get("offre", {})).lower()

        if q_lower in name or q_lower in domain or q_lower in offer:
            results.append({
                "name": record.get("entity", {}).get("name", ""),
                "domain": record.get("source", {}).get("canonical_domain", ""),
                "website": record.get("entity", {}).get("website", ""),
                "aio_score": record.get("aoi_readiness", {}).get("estimated_aio_score", 0),
                "asr_status": record.get("asr_derived", {}).get("asr_status", ""),
            })

    return {
        "query": q,
        "count": len(results),
        "results": sorted(results, key=lambda x: x["aio_score"], reverse=True),
    }


@app.get("/entity/{domain}")
def get_entity(domain: str):
    path = os.path.join(DATA_FOLDER, f"{domain}.json")
    if not os.path.exists(path):
        raise HTTPException(status_code=404, detail="Entity not found")

    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)


@app.get("/asr/{domain}")
def get_asr(domain: str):
    path = os.path.join(DATA_FOLDER, f"{domain}.json")
    if not os.path.exists(path):
        raise HTTPException(status_code=404, detail="ASR not found")

    with open(path, "r", encoding="utf-8") as f:
        record = json.load(f)

    return {
        "entity": record.get("entity", {}),
        "aio_blocks": record.get("aio_blocks", {}),
        "aoi_readiness": record.get("aoi_readiness", {}),
        "asr_derived": record.get("asr_derived", {}),
    }
